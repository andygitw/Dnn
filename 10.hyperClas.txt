import tensorflow as tf
from tensorflow.keras import layers, models
from tensorflow.keras.optimizers import SGD
from tensorflow.keras.initializers import HeNormal, GlorotUniform, Zeros, Ones
from tensorflow.keras import regularizers

# Load the Fashion MNIST dataset
(x_train, y_train), (x_test, y_test) = tf.keras.datasets.fashion_mnist.load_data()

# Normalize the images to [0, 1]
x_train, x_test = x_train / 255.0, x_test / 255.0

def build_model(units, lr, dropout_rate, reg_rate, momentum, activation, kernel_init, bias_init):
    model = models.Sequential([
        layers.Flatten(input_shape=(28, 28)),
        layers.Dense(units, activation=activation,
                     kernel_initializer=kernel_init,
                     bias_initializer=bias_init,
                     kernel_regularizer=regularizers.l2(reg_rate)),
        layers.Dropout(dropout_rate),
        layers.Dense(10, activation='softmax',
                     kernel_initializer=kernel_init,
                     bias_initializer=bias_init,
                     kernel_regularizer=regularizers.l2(reg_rate))
    ])

    optimizer = SGD(learning_rate=lr, momentum=momentum)
    model.compile(optimizer=optimizer,
                  loss='sparse_categorical_crossentropy',
                  metrics=['accuracy'])
    return model

# Hyperparameters to test
hidden_units = [64, 128]
learning_rates = [0.001, 0.01]
dropout_rates = [0.2, 0.5]
regularization_rates = [0.01, 0.1]
momentum_values = [0.9, 0.99]
activations = ['relu', 'tanh']
kernel_initializers = [HeNormal(), GlorotUniform()]
bias_initializers = [Zeros(), Ones()]

best_acc = 0
best_params = None

# Grid search over combinations
for units in hidden_units:
    for lr in learning_rates:
        for dropout_rate in dropout_rates:
            for reg_rate in regularization_rates:
                for momentum in momentum_values:
                    for activation in activations:
                        for kernel_init in kernel_initializers:
                            for bias_init in bias_initializers:
                                print(f"Training with units={units}, lr={lr}, dropout={dropout_rate}, "
                                      f"reg={reg_rate}, momentum={momentum}, activation={activation}, "
                                      f"kernel_init={type(kernel_init).__name__}, bias_init={type(bias_init).__name__}")
                                model = build_model(units, lr, dropout_rate, reg_rate, momentum,
                                                    activation, kernel_init, bias_init)
                                
                                model.fit(x_train, y_train, epochs=3, verbose=0)
                                test_loss, test_acc = model.evaluate(x_test, y_test, verbose=0)
                                print(f"Test Accuracy: {test_acc:.4f}\n")
                                
                                if test_acc > best_acc:
                                    best_acc = test_acc
                                    best_params = (units, lr, dropout_rate, reg_rate, momentum,
                                                   activation, type(kernel_init).__name__, type(bias_init).__name__)

# Final best result
print(f"Best Hyperparameters:\n"
      f"  Hidden Units = {best_params[0]}\n"
      f"  Learning Rate = {best_params[1]}\n"
      f"  Dropout Rate = {best_params[2]}\n"
      f"  L2 Regularization Rate = {best_params[3]}\n"
      f"  Momentum = {best_params[4]}\n"
      f"  Activation Function = {best_params[5]}\n"
      f"  Kernel Initializer = {best_params[6]}\n"
      f"  Bias Initializer = {best_params[7]}")
print(f"Best Accuracy Achieved: {best_acc:.4f}")
